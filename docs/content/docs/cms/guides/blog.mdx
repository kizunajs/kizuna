1. An `Articles` collection.
2. `/blog/[slug]` shows one.
3. `/blog` lists them.
4. Metadata and the sitemap.

## The collection

```ts title="src/cms/content.ts"
export const ArticleId = Kizuna.brand('ArticleId', z.string());

export const Articles = defineCollection({
    name: 'articles',
    id: ArticleId,
    fields: [
        {
            name: 'title',
            schema: z.string().min(1).max(80),
        },
        {
            name: 'slug',
            schema: z.string().min(1).max(80),
        },
        {
            name: 'summary',
            schema: z.string().max(200),
        },
        {
            name: 'cover',
            schema: ImageSchema,
        },
        {
            name: 'author',
            schema: Employees.id,
        },
        {
            name: 'body',
            schema: RichTextSchema,
        },
        {
            name: 'publishedAt',
            schema: z.iso.date(),
        },
    ],
    indexes: ['publishedAt'],
});
```

## One article

```ts title="src/app/blog/[slug]/content.ts"
export default definePage({
    name: 'articlePage',
    collection: Articles,
    metadata: {
        title: ({ content }) => content.title,
        description: ({ content }) => content.summary,
        image: ({ content }) => content.cover,
        type: 'Article',
    },
});
```

```tsx title="src/app/blog/[slug]/page.tsx"
export const generateMetadata = kizuna.content.pages.articlePage.metadata;

export default async function ArticlePage({ params }: { params: Promise<{ slug: string }> }) {
    const article = await kizuna.content.pages.articlePage.get(await params);
    const author = await kizuna.content.collections.employees.get({
        id: article.author,
    });

    return (
        <article>
            <JsonLd type="Article" of={article} />
            <h1>{article.title}</h1>
            <p>{author.name}</p>
            <RichText value={article.body} />
        </article>
    );
}
```

## The list

```tsx title="src/app/blog/page.tsx"
const { items } = await kizuna.content.collections.articles.list({
    select: ['title', 'slug', 'summary', 'cover'],
    orderBy: 'publishedAt',
    limit: 20,
});
```
