## The offices and the team

```tsx title="src/app/offices/page.tsx"
export default async function OfficesPage() {
    const { items: offices } = await kizuna.content.collections.offices.list();
    const { items: employees } = await kizuna.content.collections.employees.list();

    return offices.map((office) => (
        <section key={office.id}>
            <JsonLd type="LocalBusiness" of={office} />
            <h2>{office.name}</h2>
            <p>{office.address}</p>
            <ul>
                {employees
                    .filter((employee) => employee.office === office.id)
                    .map((employee) => (
                        <li key={employee.id}>{employee.name}</li>
                    ))}
            </ul>
        </section>
    ));
}
```

## The contact form

The page's words are content. The form posts to a route on the app API, which is not:

```ts title="src/app/contact/content.ts"
export default definePage({
    name: 'contactPage',
    fields: [
        {
            name: 'heading',
            schema: z.string().max(60),
        },
        {
            name: 'intro',
            schema: z.string().max(300),
        },
    ],
});
```

```ts title="src/routes/contact.ts"
submitContact: k
    .route({
        method: 'POST',
        path: '/contact',
        auth: false,
        body: z.object({
            name: z.string().max(80),
            email: z.email(),
            message: z.string().max(2000),
        }),
        responses: {
            204: z.void(),
        },
    })
    .handler(async ({ body, plugins }) => {
        await plugins.resend.send({
            to: 'hello@example.com',
            subject: `Message from ${body.name}`,
            text: body.message,
        });

        return {
            status: 204,
        };
    }),
```

The page shows the words and the form. A server action calls the route:

```tsx title="src/app/contact/page.tsx"
export default async function ContactPage() {
    const content = await kizuna.content.pages.contactPage.get();

    return (
        <main>
            <h1>{content.heading}</h1>
            <p>{content.intro}</p>
            <form action={sendMessage}>
                <input name="name" required maxLength={80} />
                <input name="email" type="email" required />
                <textarea name="message" required maxLength={2000} />
                <button type="submit">Send</button>
            </form>
        </main>
    );
}
```

```ts title="src/app/contact/actions.ts"
'use server';

import { redirect } from 'next/navigation';
import { apiClient } from '../../lib/api-client';

export async function sendMessage(formData: FormData) {
    await apiClient.contact.submitContact({
        body: {
            name: String(formData.get('name')),
            email: String(formData.get('email')),
            message: String(formData.get('message')),
        },
    });
    redirect('/contact/thanks');
}
```
